## script.js
```javascript
// Configuration: change these to modify behavior or texts
const NUM_CARDS = 5; // <- change number of cards here (n)
const MAX_COUNT = 3;  // <- change max counter per card here (max)

// Cards data: replace the Arabic placeholder text below with your Athkar lines.
// Provide exactly NUM_CARDS objects. Each object has `front` and `back` strings.
const cardsData = Array.from({length: NUM_CARDS}, (_,i) => ({
	front: `الذكر الأمامي - بطاقة ${i+1}`, // <-- change front text here
	back:  `لا اله الا الله ${i+1}`  // <-- change back text here
}));

// Counters per card (will be loaded from localStorage if available)
let counters = new Array(NUM_CARDS).fill(0);

// App state
let currentIndex = 0;
let isFlipped = false;
let hasStarted = false;

const cardOuter = document.getElementById('card');
const cardIndexEl = document.getElementById('cardIndex');
const overlay = document.getElementById('overlay');

// LocalStorage keys and helpers
const LS_KEY_COUNTERS = 'athkar_counters_v1';
const LS_KEY_INDEX = 'athkar_index_v1';

function loadState(){
	try{
		const raw = localStorage.getItem(LS_KEY_COUNTERS);
		if(raw){
			const parsed = JSON.parse(raw);
			if(Array.isArray(parsed)){
				// ensure length matches
				counters = new Array(NUM_CARDS).fill(0);
				for(let i=0;i<Math.min(parsed.length,NUM_CARDS);i++) counters[i]=Number(parsed[i])||0;
			}
		}
		const idxRaw = localStorage.getItem(LS_KEY_INDEX);
		if(idxRaw!=null){
			const n = Number(idxRaw);
			if(!isNaN(n) && n>=0 && n<NUM_CARDS) currentIndex = n;
		}
	}catch(e){ console.warn('Failed loading state',e); }
}

function saveState(){
	try{
		localStorage.setItem(LS_KEY_COUNTERS, JSON.stringify(counters));
		localStorage.setItem(LS_KEY_INDEX, String(currentIndex));
	}catch(e){ console.warn('Failed saving state',e); }
}

// Create a card DOM element for index
function createCardElement(index, flipped){
	// wrapper handles vertical sliding
	const wrapper = document.createElement('div');
	wrapper.className = 'card-wrap';

	const container = document.createElement('div');
	container.className = 'card' + (flipped? ' flipped':'');

	const front = document.createElement('div');
	front.className = 'face front';
	const frontText = document.createElement('div');
	frontText.className = 'text-large';
	frontText.textContent = cardsData[index].front;
	front.appendChild(frontText);

	const back = document.createElement('div');
	back.className = 'face back';
	const backText = document.createElement('div');
	backText.className = 'text-large';
	backText.textContent = cardsData[index].back;
	back.appendChild(backText);

	// Controls stay fixed and don't flip with card
	const controls = document.createElement('div');
	controls.className = 'controls';

	const counterWrap = document.createElement('div');
	counterWrap.className = 'counter';

	// SVG circle progress (larger)
	const svgNS = 'http://www.w3.org/2000/svg';
	const svg = document.createElementNS(svgNS, 'svg');
	svg.setAttribute('viewBox','0 0 72 72');
	const bg = document.createElementNS(svgNS, 'circle');
	bg.setAttribute('cx','36'); bg.setAttribute('cy','36'); bg.setAttribute('r','29'); bg.setAttribute('class','bg-circle');
	const prog = document.createElementNS(svgNS, 'circle');
	prog.setAttribute('cx','36'); prog.setAttribute('cy','36'); prog.setAttribute('r','29'); prog.setAttribute('class','progress-circle');
	svg.appendChild(bg); svg.appendChild(prog);

	const circleWrap = document.createElement('div');
	circleWrap.className = 'counter-circle';
	circleWrap.id = 'counterCircle';
	circleWrap.appendChild(svg);

	const inner = document.createElement('div');
	inner.className = 'inner';
	const val = document.createElement('div');
	val.className = 'value';
	val.id = 'counterValue';
	val.textContent = counters[index];
	inner.appendChild(val);
	circleWrap.appendChild(inner);

	// Reset button as icon
	const resetBtn = document.createElement('button');
	resetBtn.className = 'btn reset';
	resetBtn.innerHTML = '⟲';
	resetBtn.title = 'Reset counter';
	resetBtn.onclick = (ev)=>{ ev.stopPropagation(); resetCounter(); };

	counterWrap.appendChild(circleWrap);
	controls.appendChild(counterWrap);
	controls.appendChild(resetBtn);

	container.appendChild(front);
	container.appendChild(back);
	wrapper.appendChild(container);
	wrapper.appendChild(controls); // Controls outside the flipping card

	// clicking/tapping whole wrapper increments (unless it was a drag)
	wrapper.addEventListener('click', ()=>{ if(movedDuringTouch) return; incrementCounter(); });

	// Set initial fill (SVG)
	updateCircle(circleWrap, index);

	return wrapper;
}

function updateCircle(circleEl, index){
	if(!circleEl) return;
	const valn = counters[index]||0;
	const denom = Math.max(1, MAX_COUNT);
	const ratio = Math.max(0, Math.min(1, valn/denom));
	const svg = circleEl.querySelector('svg');
	if(!svg) return;
	const prog = svg.querySelector('.progress-circle');
	const radius = 29; // Updated radius
	const circumference = 2*Math.PI*radius;
	prog.style.strokeDasharray = `${circumference}`;
	const offset = circumference * (1 - ratio);
	prog.style.strokeDashoffset = `${offset}`;
	// update numeric value inside
	const valEl = circleEl.querySelector('#counterValue');
	if(valEl) valEl.textContent = valn;
}

// Render initial card (no animation)
function render(){
	cardOuter.innerHTML = '';
	isFlipped = false;
	const el = createCardElement(currentIndex, isFlipped);
	cardOuter.appendChild(el);
	// mark as active wrapper
	el.id = 'activeCard';
	updateFooter();
	updateIncState();
}

function updateFooter(){
	const progressPct = NUM_CARDS > 1 ? ((currentIndex) / (NUM_CARDS - 1)) * 100 : 100;
	cardIndexEl.innerHTML = `
		<div class="progress-bar">
			<div class="progress-fill" style="width: ${progressPct}%"></div>
		</div>
		<div class="card-info">البطاقة ${currentIndex+1} من ${NUM_CARDS}</div>
	`;
}

function incrementCounter(){
	if(counters[currentIndex] >= MAX_COUNT) return;
	counters[currentIndex]++;
	const circle = document.getElementById('counterCircle');
	updateCircle(circle, currentIndex);
	updateIncState();
	saveState();
}

function resetCounter(){
	counters[currentIndex] = 0;
	const circle = document.getElementById('counterCircle');
	updateCircle(circle, currentIndex);
	updateIncState();
	saveState();
}

function updateIncState(){
	const resetBtn = document.querySelector('.btn.reset');
	if(resetBtn) resetBtn.disabled = counters[currentIndex] <= 0;
}

// Navigation with animation
function goToIndex(idx){
	if(idx < 0) idx = 0;
	if(idx >= NUM_CARDS) idx = NUM_CARDS - 1;
	if(idx === currentIndex) return;
	const direction = (idx > currentIndex) ? 'up' : 'down';
	animateToIndex(idx, direction);
}

function animateToIndex(newIndex, direction){
	const currentWrap = document.getElementById('activeCard');
	const newWrap = createCardElement(newIndex, false);
  
	// Ensure both wrappers are positioned absolutely within the same container
	newWrap.style.transform = `translateY(${direction==='up'? '100%':'-100%'})`;
	newWrap.style.opacity = '0';
	cardOuter.appendChild(newWrap);

	// Force layout calculation
	newWrap.offsetHeight;

	// Start animations
	requestAnimationFrame(() => {
		if(currentWrap){
			currentWrap.style.transform = `translateY(${direction==='up'? '-100%':'100%'})`;
			currentWrap.style.opacity = '0';
		}
		newWrap.style.transform = 'translateY(0)';
		newWrap.style.opacity = '1';
	});

	// Cleanup after animation completes
	const cleanup = ()=>{
		if(currentWrap && currentWrap.parentNode) currentWrap.parentNode.removeChild(currentWrap);
		newWrap.id = 'activeCard';
		isFlipped = false;
		currentIndex = newIndex;
		updateFooter();
		updateIncState();
		saveState();
	};

	setTimeout(cleanup, 450); // Match CSS transition timing
}

function toggleFlip(){
	isFlipped = !isFlipped;
	const wrap = document.getElementById('activeCard');
	if(!wrap) return;
	const card = wrap.querySelector('.card');
	const controls = wrap.querySelector('.controls');
	if(card){
		if(isFlipped){
			card.classList.add('flipped');
			if(controls) controls.style.display = 'none';
		} else {
			card.classList.remove('flipped');
			if(controls) controls.style.display = 'flex';
		}
	}
}

// Swipe handling (touch + mouse) with drag detection to avoid accidental increments
let startX = 0, startY = 0, isTouching = false;
let movedDuringTouch = false;
const H_THRESH = 40; // px horizontal threshold
const V_THRESH = 40; // px vertical threshold

function onStart(x,y){ startX = x; startY = y; isTouching = true; movedDuringTouch = false; }
function onMove(x,y){
	if(!isTouching) return;
	if(Math.abs(x - startX) > 8 || Math.abs(y - startY) > 8) movedDuringTouch = true;
}
function onEnd(x,y){
	if(!isTouching) return; isTouching = false;
	const dx = x - startX; const dy = y - startY;
	if(Math.abs(dx) > Math.abs(dy) && Math.abs(dx) > H_THRESH){
		// horizontal swipe -> flip
		toggleFlip();
	} else if(Math.abs(dy) > Math.abs(dx) && Math.abs(dy) > V_THRESH){
		// vertical swipe -> change card index
		if(dy < 0) goToIndex(currentIndex + 1); // swipe up -> next
		else goToIndex(currentIndex - 1); // swipe down -> prev
	}
}

// Touch events
cardOuter.addEventListener('touchstart', (e)=>{ const t=e.changedTouches[0]; onStart(t.clientX,t.clientY); });
cardOuter.addEventListener('touchmove', (e)=>{ const t=e.changedTouches[0]; onMove(t.clientX,t.clientY); e.preventDefault(); });
cardOuter.addEventListener('touchend', (e)=>{ const t=e.changedTouches[0]; onEnd(t.clientX,t.clientY); });

// Mouse events (for desktop testing)
let mouseDown = false;
cardOuter.addEventListener('mousedown', (e)=>{ mouseDown=true; onStart(e.clientX,e.clientY); });
window.addEventListener('mousemove', (e)=>{ if(mouseDown) onMove(e.clientX,e.clientY); });
window.addEventListener('mouseup', (e)=>{ if(mouseDown){ mouseDown=false; onEnd(e.clientX,e.clientY); }});

// Keyboard support
window.addEventListener('keydown', (e)=>{
	if(e.key === 'ArrowUp') goToIndex(currentIndex + 1);
	if(e.key === 'ArrowDown') goToIndex(currentIndex - 1);
	if(e.key === 'ArrowLeft' || e.key === 'ArrowRight') toggleFlip();
});

// Initialize
loadState();
render();

// Hide overlay on first interaction
function hideOverlay(){
	if(!hasStarted && overlay){
		hasStarted = true;
		overlay.classList.add('hidden');
	}
}

// Expose some helpers for debugging in console
window.__ATHK = { goToIndex, toggleFlip, cardsData, counters, saveState, loadState };

// Setup cloud animation
function setupCloudCanvas(){
	const canvas = document.createElement('canvas');
	canvas.id = 'cloudCanvas';
	canvas.width = window.innerWidth;
	canvas.height = window.innerHeight;
	document.body.insertBefore(canvas, document.body.firstChild);
  
	const ctx = canvas.getContext('2d');
	let time = 0;
  
	// Cloud objects with different speeds and positions
	const clouds = [
		{ x: 50, y: 80, size: 80, speed: 0.0003, offset: 0 },
		{ x: 300, y: 120, size: 60, speed: 0.00025, offset: Math.PI },
		{ x: 600, y: 100, size: 90, speed: 0.00035, offset: Math.PI * 0.5 },
		{ x: 900, y: 140, size: 70, speed: 0.0002, offset: Math.PI * 1.5 },
		{ x: 1200, y: 110, size: 75, speed: 0.00028, offset: Math.PI * 0.25 }
	];
  
	function drawClouds(){
		ctx.clearRect(0, 0, canvas.width, canvas.height);
		time++;
    
		clouds.forEach(cloud => {
			const offsetX = Math.sin(time * cloud.speed + cloud.offset) * 150;
			drawRealisticCloud(ctx, cloud.x + offsetX, cloud.y, cloud.size);
		});
    
		requestAnimationFrame(drawClouds);
	}
  
	function drawRealisticCloud(ctx, x, y, size){
		ctx.fillStyle = 'rgba(255, 255, 255, 0.35)';
    
		// Multiple overlapping circles to create realistic cloud shape
		ctx.beginPath();
		ctx.arc(x - size * 0.35, y, size * 0.45, 0, Math.PI * 2);
		ctx.fill();
    
		ctx.beginPath();
		ctx.arc(x - size * 0.05, y - size * 0.15, size * 0.55, 0, Math.PI * 2);
		ctx.fill();
    
		ctx.beginPath();
		ctx.arc(x + size * 0.3, y - size * 0.05, size * 0.5, 0, Math.PI * 2);
		ctx.fill();
    
		ctx.beginPath();
		ctx.arc(x + size * 0.55, y + size * 0.1, size * 0.4, 0, Math.PI * 2);
		ctx.fill();
    
		// Subtle shadow effect at bottom of cloud
		const gradient = ctx.createLinearGradient(x, y - size * 0.2, x, y + size * 0.3);
		gradient.addColorStop(0, 'rgba(200,220,240,0)');
		gradient.addColorStop(1, 'rgba(150,180,220,0.1)');
		ctx.fillStyle = gradient;
		ctx.fillRect(x - size * 0.6, y, size * 1.2, size * 0.4);
	}
  
	window.addEventListener('resize', () => {
		canvas.width = window.innerWidth;
		canvas.height = window.innerHeight;
	});
  
	drawClouds();
}

setupCloudCanvas();

// Track interactions to hide overlay
cardOuter.addEventListener('click', hideOverlay);
cardOuter.addEventListener('touchstart', hideOverlay);
```

## README.md
```markdown
Athkar Cards - Simple Web App

How to open

- Open `index.html` in a browser (double-click or use a static file server).

Where to change text and variables

- Edit `script.js`:
	- `NUM_CARDS` at the top controls how many cards are available (default 5).
	- `MAX_COUNT` controls the maximum counter per card (default 3).
	- `cardsData` array contains objects with `front` and `back` strings for each card. Replace the placeholder Arabic text with your Athkar.

Persistence (localStorage)

- The app now persists the per-card counters and the current card index in `localStorage`.
- Counters are saved under key `athkar_counters_v1` and the current index under `athkar_index_v1`.
- To reset stored progress, clear site data in browser or run `localStorage.removeItem('athkar_counters_v1')` in the console.

Interactions changes

- Tap the whole card to increase the counter (no separate + button anymore). Swiping up/down still switches cards and left/right flips.
- Each card has a `Reset` button at the bottom to reset that card's counter to 0.

Interactions

- Swipe left/right (or press left/right arrow) to flip the active card.
- Swipe up/down (or press up/down arrow) to move between cards.
- Tap the `+` button at the bottom of the card to increment the counter for that card (respects `MAX_COUNT`).

Notes

- The layout is responsive and uses CSS 3D transform for the flip animation.
- Counters are stored in memory (page reload resets them). If you want persistence, I can add localStorage support.
```
